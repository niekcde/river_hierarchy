def route(s,k,x,flow,nsub,dt):
 f=s['f'];out=s['edges'];signals={s['root']:np.repeat(flow,nsub)};steady={s['root']:float(flow[0])};mn=np.inf
 den=dt+2*k*(1-x);a=(dt-2*k*x)/den;b=(dt+2*k*x)/den;c=(2*k*(1-x)-dt)/den
 assert (abs(c)<1).all()
 v=f.v.to_numpy();count=f['count'].to_numpy(int)
 for node in s['order']:
  signal=signals.pop(node);initial=steady.pop(node)
  if node==s['end']:return signal.reshape(-1,nsub).mean(axis=1),mn,initial/float(flow[0])
  for i in out[node]:
   fraction=s['ratios'][i] if node==s['root'] else float(np.float32(round(s['ratios'][i],6)))
   value=signal*fraction;eq=initial*fraction
   for seg in range(count[i]):
    b0,b1=(a[i]+b[i],0.) if node==s['root'] and seg==0 else (a[i],b[i])
    value,_=lfilter([b0,b1],[1.,-c[i]],value,zi=[(b1+c[i])*eq]);mn=min(mn,float(value.min()))
   assert np.isfinite(value).all()
   signals[v[i]]=signals.get(v[i],0)+value;steady[v[i]]=steady.get(v[i],0)+eq
 assert False
