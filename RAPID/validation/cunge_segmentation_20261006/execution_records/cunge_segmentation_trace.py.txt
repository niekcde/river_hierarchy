def run(profile,pick=None):
 x=s['xs'][profile];k,N,dx,label=s['variants'][id(x)];den=dt+2*k*(1-x);aa=(dt-2*k*x)/den;bb=(dt+2*k*x)/den;dd=(2*k*(1-x)-dt)/den;signals={s['root']:flow};v=s['f'].v.to_numpy();worst=0;trace=None;meta=None
 for u in s['order']:
  sig=signals.pop(u)
  if u==s['end']:continue
  for i in s['edges'][u]:
   val=sig*s['ratios'][i];eq=flow[0]*s['shares'][i]
   for j in range(N[i]):val,_=lfilter([aa[i],bb[i]],[1,-dd[i]],val,zi=[(bb[i]+dd[i])*eq])
   under=(val.min()-eq)/(amp*s['shares'][i])
   if (pick is None and under<worst) or pick==i:
    worst=under;trace=(val-eq)/(amp*s['shares'][i]);meta=dict(index=int(i),id_link=int(s['f'].iloc[i].id_link),physical_length_m=float(s['f'].iloc[i].link_length_m),celerity_mps=float(s['f'].iloc[i].rapid_celerity_mps),x=float(x[i]),a=float(aa[i]),b=float(bb[i]),d=float(dd[i]),N=int(N[i]),minimum_excess_fraction=float(under))
   signals[v[i]]=signals.get(v[i],0)+val
 return trace,meta
