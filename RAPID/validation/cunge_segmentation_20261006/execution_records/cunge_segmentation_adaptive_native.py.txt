def route(s,x,dt,flow):
 k,N,dx,label=s['variants'][id(x)];den=dt+2*k*(1-x);a=(dt-2*k*x)/den;b=(dt+2*k*x)/den;d=(2*k*(1-x)-dt)/den;assert (d>=0).all() and (d<1).all()
 signals={s['root']:flow};v=s['f'].v.to_numpy();mn=np.inf;under=0.
 for u in s['order']:
  signal=signals.pop(u)
  if u==s['end']:return signal,mn,under
  for i in s['edges'][u]:
   val=signal*s['ratios'][i];eq=flow[0]*s['shares'][i]
   for _ in range(N[i]):
    val,_=lfilter([a[i],b[i]],[1.,-d[i]],val,zi=[(b[i]+d[i])*eq]);mn=min(mn,float(val.min()));under=min(under,float((val.min()-eq)/(np.ptp(flow)*s['shares'][i])))
   signals[v[i]]=signals.get(v[i],0)+val
 raise ValueError('No outlet')
