def pairs(m):
 rows=[]
 for (ex,p,dt,forcing),g in m.groupby(['example','profile','dt_s','forcing']):
  parent=g[g.state=='S000_base'].iloc[0]
  for _,r in g[g.state!='S000_base'].iterrows():rows.append(dict(example=ex,state=r.state,profile=p,dt_s=dt,forcing=forcing,delta_peak_h=r.peak_time_h-parent.peak_time_h,peak_excess_change_pct=100*(r.peak_excess_cms/parent.peak_excess_cms-1)))
 return pd.DataFrame(rows)
