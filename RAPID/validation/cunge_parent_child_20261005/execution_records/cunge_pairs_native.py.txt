def run_native(ex):
 dest=O/'native'/ex;dest.mkdir(parents=True,exist_ok=True)

 ref=grid['reference_cms'];qmid=grid['midpoint_cms'];h=pd.read_csv(src['hydrograph']);st=h.time_seconds.to_numpy()-grid['actual_start_offset_seconds'];sq=h.q_inflow_cms.to_numpy()

 wanted=[v for v in grid['sources'] if v[0]=='S000_base' or v[0] in chosen[ex]]
 rows=[]
 for name,path,fp in wanted:
  f=pd.read_parquet(path);s=exp.setup(f,ref,qmid)
  for profile in ['fixed_0.1','cunge_reference']:
   q,mn,neg,_=route(f,s['xs'][profile],dt,forcing,base)
   mu,var=exp.moments(s,s['xs'][profile],dt)
   metrics=exp.metrics(t,q,forcing,base,sq[-1],'saved_event',mu,var,0)
