def route(f,x,dt,forcing,base,verify=False):
 G=nx.DiGraph();out={}
 for i,r in enumerate(f.itertuples()):G.add_edge(r.u,r.v);out.setdefault(r.u,[]).append(i)
 roots=[v for v in G if G.in_degree(v)==0];ends=[v for v in G if G.out_degree(v)==0]
 assert len(roots)==len(ends)==1
 signals={roots[0]:forcing-base};shares={roots[0]:1.};minimum=np.inf;neg=0;idx=0;edge_states=[];edge_final=[];expanded=[]
 den=dt+2*f.rapid_k.to_numpy()*(1-x);a=(dt-2*f.rapid_k.to_numpy()*x)/den;b=(dt+2*f.rapid_k.to_numpy()*x)/den;c=(2*f.rapid_k.to_numpy()*(1-x)-dt)/den
 assert (abs(c)<1).all()
 for node in nx.topological_sort(G):
  signal=signals.pop(node);share=shares.pop(node)
  if node==ends[0]:answer=signal+base;continue
  ids=out[node];width=f.iloc[ids].width.sum()
  for i in ids:
   r=f.iloc[i];ratio=r.width/width;s=signal*ratio;flowshare=share*ratio
   u=node
   for j in range(int(r['count'])):
    num=[a[i]+b[i]] if node==roots[0] and j==0 else [a[i],b[i]]
    s=lfilter(num,[1,-c[i]],s)
    mn=float(s.min()+base*flowshare);minimum=min(minimum,mn);neg+=int(np.count_nonzero(s+base*flowshare < -1e-10*base))
    if verify:
     v=int(r.v) if j==r['count']-1 else -1000000-idx
     expanded.append((u,v,r.width,r.rapid_k,x[i],flowshare));u=v;idx+=1
     edge_final.append(s[-1])
   signals[r.v]=signals.get(r.v,0)+s;shares[r.v]=shares.get(r.v,0)+flowshare
 error=None
 if verify:
  net=np.zeros((len(expanded),len(expanded)))
  for i,(_,v,_,_,_,_) in enumerate(expanded):
   ids=[j for j,e in enumerate(expanded) if e[0]==v]
   for j in ids:net[j,i]=expanded[j][2]/sum(expanded[k][2] for k in ids)
  mats=rte_mat(csc_matrix(net),*ccc_mat(np.array([e[3] for e in expanded]),np.array([e[4] for e in expanded]),dt))
  solve=factorized(mats[0]);state=np.zeros(len(expanded));q=[]
  for value in forcing-base:
   ext=np.array([value*e[5] if e[0]==roots[0] else 0. for e in expanded])
   _,state=mus_rte(*mats,1,state,ext,solve)
   q.append(sum(state[j] for j,e in enumerate(expanded) if e[1]==ends[0])+base)
  error=max(float(np.max(abs(np.array(q)-answer))),float(np.max(abs(state-edge_final))))
  assert error<1e-8*max(forcing),error
 return answer,minimum,neg,error
